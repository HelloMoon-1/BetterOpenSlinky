cmake_minimum_required(VERSION 3.24)
project(SlinkyNativeReconstruction LANGUAGES CXX)

if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "This reconstruction requires a 64-bit target.")
endif()
if(NOT WIN32 AND NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "Supported targets are Windows x64 and Linux x64.")
endif()
if(WIN32)
  enable_language(RC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(PAYLOAD_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../build" CACHE PATH "Directory with rebuilt payload-101.jar through payload-103.jar")
set(JAVA_HOME "$ENV{JAVA_HOME}" CACHE PATH "JDK installation providing JNI/JVMTI headers")
find_path(JNI_INCLUDE_DIR jni.h HINTS "${JAVA_HOME}/include" REQUIRED)
if(WIN32)
  set(JNI_PLATFORM win32)
else()
  set(JNI_PLATFORM linux)
endif()
find_path(JNI_PLATFORM_INCLUDE_DIR jni_md.h HINTS "${JAVA_HOME}/include/${JNI_PLATFORM}" REQUIRED)
foreach(id IN ITEMS 101 102 103)
  if(NOT EXISTS "${PAYLOAD_DIR}/payload-${id}.jar")
    message(FATAL_ERROR "Missing ${PAYLOAD_DIR}/payload-${id}.jar; build the Java payloads first.")
  endif()
endforeach()
if(WIN32)
file(TO_CMAKE_PATH "${PAYLOAD_DIR}" PAYLOAD_DIR_RC)
configure_file(payloads.rc.in payloads.rc @ONLY)
add_library(SlinkyNative SHARED payload.cpp "${CMAKE_CURRENT_BINARY_DIR}/payloads.rc")
else()
  add_library(SlinkyNative SHARED payload.cpp)
  find_package(Threads REQUIRED)
  target_link_libraries(SlinkyNative PRIVATE Threads::Threads ${CMAKE_DL_LIBS})
  target_compile_options(SlinkyNative PRIVATE -Wall -Wextra -Wpedantic)
  foreach(id IN ITEMS 101 102 103)
    add_custom_command(TARGET SlinkyNative POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
      "${PAYLOAD_DIR}/payload-${id}.jar" "$<TARGET_FILE_DIR:SlinkyNative>/payload-${id}.jar")
  endforeach()
endif()
target_include_directories(SlinkyNative PRIVATE "${JNI_INCLUDE_DIR}" "${JNI_PLATFORM_INCLUDE_DIR}")
if(WIN32)
target_compile_definitions(SlinkyNative PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
set_source_files_properties("${CMAKE_CURRENT_BINARY_DIR}/payloads.rc" PROPERTIES OBJECT_DEPENDS "${PAYLOAD_DIR}/payload-101.jar;${PAYLOAD_DIR}/payload-102.jar;${PAYLOAD_DIR}/payload-103.jar")
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/launcher-$<CONFIG>.rc" CONTENT "#include <windows.h>\n200 RCDATA \"$<TARGET_FILE:SlinkyNative>\"\n1 RT_MANIFEST \"${CMAKE_CURRENT_SOURCE_DIR}/launcher.manifest\"\n")
add_executable(Slinky WIN32 launcher.cpp "${CMAKE_CURRENT_BINARY_DIR}/launcher-$<CONFIG>.rc")
add_dependencies(Slinky SlinkyNative)
target_compile_definitions(Slinky PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(Slinky PRIVATE user32 gdi32)
if(MSVC)
  target_compile_options(SlinkyNative PRIVATE /W4 /utf-8 /permissive-)
  target_compile_options(Slinky PRIVATE /W4 /utf-8 /permissive-)
  target_link_options(Slinky PRIVATE /MANIFEST:NO)
endif()
endif()
